// ================== CREATE ==================
export const createTicket = async (req, res) => {
  try {
    if (!req.file) {
      return res.send("Chưa chọn ảnh");
    }

    const {
      eventName,
      holderName,
      category,
      quantity,
      pricePerTicket,
      eventDate,
      status
    } = req.body;

    const q = Number(quantity);
    const p = Number(pricePerTicket);

    if (q <= 0 || p <= 0) {
      return res.send("Quantity & price phải > 0");
    }

    const id = uuidv4();
    const imageKey = `tickets/${id}-${req.file.originalname}`;

    // ===== UPLOAD S3 =====
    await s3.send(
      new PutObjectCommand({
        Bucket: process.env.BUCKET_NAME,
        Key: imageKey,
        Body: req.file.buffer,
        ContentType: req.file.mimetype
      })
    );

    const imageUrl = `https://${process.env.BUCKET_NAME}.s3.${process.env.AWS_REGION}.amazonaws.com/${imageKey}`;

    // ===== BUSINESS =====
    const totalAmount = q * p;
    let finalAmount = totalAmount;
    let discountLabel = "Không giảm giá";

    if (category === "VIP" && q >= 4) {
      finalAmount *= 0.9;
      discountLabel = "Được giảm giá";
    } else if (category === "VVIP" && q >= 2) {
      finalAmount *= 0.85;
      discountLabel = "Được giảm giá";
    }

    await docClient.send(
      new PutCommand({
        TableName: "EventTickets",
        Item: {
          ticketId: id,
          eventName,
          holderName,
          category,
          quantity: q,
          pricePerTicket: p,
          eventDate,
          status,
          imageUrl,
          totalAmount,
          finalAmount,
          discountLabel
        }
      })
    );

    res.redirect("/");
  } catch (err) {
    console.error(err);
    res.send("Error creating ticket");
  }
};

// ================== UPDATE ==================
export const updateTicket = async (req, res) => {
  try {
    const { id } = req.params;
    const { eventName, holderName, category, quantity, pricePerTicket, eventDate, status } = req.body;

    const q = Number(quantity);
    const p = Number(pricePerTicket);

    const totalAmount = q * p;
    let finalAmount = totalAmount;
    let discountLabel = "Không giảm giá";

    if (category === "VIP" && q >= 4) {
      finalAmount *= 0.9;
      discountLabel = "Được giảm giá";
    } else if (category === "VVIP" && q >= 2) {
      finalAmount *= 0.85;
      discountLabel = "Được giảm giá";
    }

    let updateExp = "SET eventName=:e, holderName=:h, category=:c, quantity=:q, pricePerTicket=:p, eventDate=:d, #st=:s, totalAmount=:t, finalAmount=:f, discountLabel=:dl";

    let attrValues = {
      ":e": eventName,
      ":h": holderName,
      ":c": category,
      ":q": q,
      ":p": p,
      ":d": eventDate,
      ":s": status,
      ":t": totalAmount,
      ":f": finalAmount,
      ":dl": discountLabel
    };

    let attrNames = { "#st": "status" };

    // ===== nếu có ảnh mới =====
    if (req.file) {
      const imageKey = `tickets/${id}-${req.file.originalname}`;

      await s3.send(
        new PutObjectCommand({
          Bucket: process.env.BUCKET_NAME,
          Key: imageKey,
          Body: req.file.buffer,
          ContentType: req.file.mimetype
        })
      );

      const imageUrl = `https://${process.env.BUCKET_NAME}.s3.${process.env.AWS_REGION}.amazonaws.com/${imageKey}`;

      updateExp += ", imageUrl=:img";
      attrValues[":img"] = imageUrl;
    }

    await docClient.send(
      new UpdateCommand({
        TableName: "EventTickets",
        Key: { ticketId: id },
        UpdateExpression: updateExp,
        ExpressionAttributeValues: attrValues,
        ExpressionAttributeNames: attrNames
      })
    );

    res.redirect("/");
  } catch (err) {
    console.error(err);
    res.send("Error updating ticket");
  }
};

// ================== DELETE ==================
export const deleteTicket = async (req, res) => {
  try {
    const id = req.params.id;

    const data = await docClient.send(
      new GetCommand({
        TableName: "EventTickets",
        Key: { ticketId: id }
      })
    );

    // xoá ảnh S3
    if (data.Item?.imageUrl) {
      const key = data.Item.imageUrl.split(".com/")[1];

      await s3.send(
        new DeleteObjectCommand({
          Bucket: process.env.BUCKET_NAME,
          Key: key
        })
      );
    }

    await docClient.send(
      new DeleteCommand({
        TableName: "EventTickets",
        Key: { ticketId: id }
      })
    );

    res.redirect("/");
  } catch (err) {
    console.error(err);
    res.send("Error deleting ticket");
  }
};

// ================== SEARCH ==================
export const searchTicket = async (req, res) => {
  const keyword = req.query.keyword?.toLowerCase() || "";

  const data = await docClient.send(
    new ScanCommand({ TableName: "EventTickets" })
  );

  const filtered = data.Items.filter(t =>
    t.eventName.toLowerCase().includes(keyword) ||
    t.holderName.toLowerCase().includes(keyword)
  );

  res.render("index", { tickets: filtered });
};

// ================== FILTER ==================
export const filterByStatus = async (req, res) => {
  const status = req.query.status;

  const data = await docClient.send(
    new ScanCommand({ TableName: "EventTickets" })
  );

  const filtered = data.Items.filter(t => t.status === status);

  res.render("index", { tickets: filtered });
};
